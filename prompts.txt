create table users(
    id int primary key auto_increment, 
    name varchar(100),
    emailorcontact varchar(100), 
    identity_proof varchar(255),
    profile_photo varchar(255),
    bank_account varchar(20),
    ifsc varchar(20),
    cheque varchar(255),
    users_type varchar(20),
    password varchar(100)
);
CREATE TABLE hospitals (
    id INT AUTO_INCREMENT PRIMARY KEY,
    users_id INT NOT NULL,
    hospital_images JSON,
    hospital_name VARCHAR(255) NOT NULL,
    address TEXT NOT NULL,
    location VARCHAR(255),
    facilities TEXT,
    rooms JSON,
    doctors JSON,
    hospital_reg_certificate VARCHAR(255),
    shop_license VARCHAR(255),
    medical_council_registration VARCHAR(255),
    electricity_bill VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_hospital_users
    FOREIGN KEY (users_id) REFERENCES users(id)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);
create table ambulances(
    id int auto_increment primary key, 
    users_id int not null,
    ambulance_type varchar(100),
    base_chrge varchar(100),
    min_chrge varchar(100),
    night_chrg varchar(100),
    wait_chrg varchar(100),
    status varchar(100),
    eta varchar(100),
    book_time_slot varchar(255),
    area varchar(255),
    description varchar(255),
    lic varchar(255),
    rc varchar(255),
    driver_exp varchar(10),
    veh_ins varchar(255),
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT `fk_ambulance_users`
    FOREIGN KEY (`users_id`) REFERENCES `users`(`id`)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);

    pathologist json,
    adv_equipment varchar(5),
    lab_hrs varchar(100),
